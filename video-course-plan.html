Current State Analysis

What Already Exists

  • CLASS event type with classDays, classLevel (Beginner/Intermediate/Advanced)
  • Categories: Steppin, Line Dancing, Walking
  • Image upload infrastructure (Convex storage, compression, drag & drop)
  • Instructor (Organizer) role with dashboard and management
  • Payment system: Stripe Connect, PayPal, credits system
  • Subscription model: Tiered plans already built for restaurants (reusable pattern)
  • Enrollment/ticketing for classes

What's Missing for Video Classes

  • Video upload/storage infrastructure
  • Video player component
  • Course/curriculum structure (modules, lessons)
  • Progress tracking for students
  • Video-specific schema fields
  • Subscription/membership for content access

Tier 3: Database Schema

Add these 7 new tables to convex/schema.ts:

1. Video Courses - Main course container

videoCourses: defineTable({
  instructorId: v.id("users"),
  title: v.string(),
  slug: v.string(),                              // URL-friendly slug
  description: v.string(),
  shortDescription: v.optional(v.string()),      // For cards
  category: v.union(v.literal("Steppin"), v.literal("Line Dancing"), v.literal("Walking")),
  level: v.union(v.literal("Beginner"), v.literal("Intermediate"), v.literal("Advanced")),
  thumbnailStorageId: v.optional(v.id("_storage")),
  thumbnailUrl: v.optional(v.string()),
  promoVideoUrl: v.optional(v.string()),         // Free preview video

  // Pricing
  priceCents: v.number(),                        // 0 = free, else one-time price
  hasSubscription: v.boolean(),                  // Whether subscription access available

  // Status
  status: v.union(v.literal("draft"), v.literal("published"), v.literal("archived")),

  // Computed/cached
  totalDurationSeconds: v.optional(v.number()),
  lessonCount: v.optional(v.number()),
  enrollmentCount: v.optional(v.number()),
  averageRating: v.optional(v.number()),

  createdAt: v.number(),
  updatedAt: v.number(),
  publishedAt: v.optional(v.number()),
})
.index("by_instructor", ["instructorId"])
.index("by_slug", ["slug"])
.index("by_category", ["category"])
.index("by_status", ["status"])

2. Course Modules - Chapter groupings

courseModules: defineTable({
  courseId: v.id("videoCourses"),
  title: v.string(),
  description: v.optional(v.string()),
  order: v.number(),
  createdAt: v.number(),
})
.index("by_course", ["courseId"])

3. Video Lessons - Individual videos

videoLessons: defineTable({
  courseId: v.id("videoCourses"),
  moduleId: v.optional(v.id("courseModules")),
  title: v.string(),
  description: v.optional(v.string()),

  // Video data (Mux)
  muxAssetId: v.optional(v.string()),
  muxPlaybackId: v.optional(v.string()),
  muxUploadId: v.optional(v.string()),           // For tracking upload
  processingStatus: v.union(
    v.literal("waiting"),
    v.literal("processing"),
    v.literal("ready"),
    v.literal("error")
  ),

  // Metadata
  durationSeconds: v.optional(v.number()),
  thumbnailUrl: v.optional(v.string()),          // Auto-generated by Mux

  // Settings
  order: v.number(),
  isFreePreview: v.boolean(),                    // Free to watch without purchase
  isPublished: v.boolean(),

  createdAt: v.number(),
  updatedAt: v.number(),
})
.index("by_course", ["courseId"])
.index("by_module", ["moduleId"])

4. Course Enrollments - Purchase records

courseEnrollments: defineTable({
  userId: v.id("users"),
  courseId: v.id("videoCourses"),

  // Access type
  accessType: v.union(
    v.literal("purchase"),                       // One-time buy
    v.literal("subscription"),                   // Via membership
    v.literal("free")                            // Free course
  ),

  // Payment
  pricePaidCents: v.optional(v.number()),
  stripePaymentIntentId: v.optional(v.string()),

  // Status
  status: v.union(
    v.literal("active"),
    v.literal("expired"),
    v.literal("refunded")
  ),

  enrolledAt: v.number(),
  expiresAt: v.optional(v.number()),             // For subscription access
})
.index("by_user", ["userId"])
.index("by_course", ["courseId"])
.index("by_user_course", ["userId", "courseId"])

5. Student Progress - Video watch tracking

studentProgress: defineTable({
  userId: v.id("users"),
  courseId: v.id("videoCourses"),
  lessonId: v.id("videoLessons"),

  watchedSeconds: v.number(),                    // Current position
  totalSeconds: v.number(),                      // Video duration
  percentComplete: v.number(),                   // 0-100

  isCompleted: v.boolean(),
  completedAt: v.optional(v.number()),
  lastWatchedAt: v.number(),
})
.index("by_user_course", ["userId", "courseId"])
.index("by_user_lesson", ["userId", "lessonId"])

6. Instructor Memberships - Subscription plans

instructorMemberships: defineTable({
  instructorId: v.id("users"),
  name: v.string(),                              // "Monthly Access", "Annual Pass"
  description: v.string(),

  priceCents: v.number(),
  interval: v.union(
    v.literal("monthly"),
    v.literal("yearly"),
    v.literal("lifetime")
  ),

  // What's included
  includesAllCourses: v.boolean(),
  includedCourseIds: v.optional(v.array(v.id("videoCourses"))),

  // Stripe
  stripePriceId: v.optional(v.string()),
  stripeProductId: v.optional(v.string()),

  isActive: v.boolean(),
  createdAt: v.number(),
})
.index("by_instructor", ["instructorId"])

7. Student Membership Subscriptions

studentMembershipSubscriptions: defineTable({
  userId: v.id("users"),
  membershipId: v.id("instructorMemberships"),
  instructorId: v.id("users"),

  status: v.union(
    v.literal("active"),
    v.literal("past_due"),
    v.literal("cancelled"),
    v.literal("expired")
  ),

  stripeSubscriptionId: v.optional(v.string()),
  currentPeriodStart: v.number(),
  currentPeriodEnd: v.number(),
  cancelledAt: v.optional(v.number()),

  createdAt: v.number(),
})
.index("by_user", ["userId"])
.index("by_instructor", ["instructorId"])

Mux Pricing Estimate

Usage Cost Example
Video encoding $0.015/min 10 hours of video = $9
Video storage $0.007/GB/mo 50GB = $3.50/mo
Video delivery $0.005/min streamed 1000 views of 10-min video = $50

Startup estimate: $50-100/month with moderate usage

Scale: Costs grow with views, not uploads - good for growth
